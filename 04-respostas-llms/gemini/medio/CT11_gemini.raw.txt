```javascript
const adminMiddleware = require('./adminMiddleware');

describe('adminMiddleware', () => {
    let req;
    let res;
    let next;

    beforeEach(() => {
        req = {};
        res = {
            status: jest.fn().mockReturnThis(),
            json: jest.fn()
        };
        next = jest.fn();
    });

    it('deve chamar next() quando o usuário estiver presente e tiver o papel de admin', () => {
        req.user = { role: 'admin' };
        
        adminMiddleware(req, res, next);
        
        expect(next).toHaveBeenCalledTimes(1);
        expect(res.status).not.toHaveBeenCalled();
        expect(res.json).not.toHaveBeenCalled();
    });

    it('deve retornar erro 403 quando o usuário estiver presente, mas não tiver o papel de admin', () => {
        req.user = { role: 'user' };
        
        adminMiddleware(req, res, next);
        
        expect(next).not.toHaveBeenCalled();
        expect(res.status).toHaveBeenCalledWith(403);
        expect(res.status).toHaveBeenCalledTimes(1);
        expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
        expect(res.json).toHaveBeenCalledTimes(1);
    });

    it('deve retornar erro 403 quando o objeto user não estiver presente na requisição', () => {
        adminMiddleware(req, res, next);
        
        expect(next).not.toHaveBeenCalled();
        expect(res.status).toHaveBeenCalledWith(403);
        expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
    });

    it('deve retornar erro 403 quando o objeto user estiver presente, mas a propriedade role não existir', () => {
        req.user = {};
        
        adminMiddleware(req, res, next);
        
        expect(next).not.toHaveBeenCalled();
        expect(res.status).toHaveBeenCalledWith(403);
        expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
    });

    it('deve retornar erro 403 quando a propriedade role for null', () => {
        req.user = { role: null };
        
        adminMiddleware(req, res, next);
        
        expect(next).not.toHaveBeenCalled();
        expect(res.status).toHaveBeenCalledWith(403);
        expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
    });
});
```
