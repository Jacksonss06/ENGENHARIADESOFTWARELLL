```javascript
const adminMiddleware = require('./adminMiddleware');

describe('adminMiddleware', () => {
  let req;
  let res;
  let next;

  beforeEach(() => {
    req = {};
    res = {
      status: jest.fn().mockReturnThis(),
      json: jest.fn()
    };
    next = jest.fn();
  });

  it('deve chamar next() se o usuário estiver presente e for admin', () => {
    req.user = { role: 'admin' };

    adminMiddleware(req, res, next);

    expect(next).toHaveBeenCalledTimes(1);
    expect(res.status).not.toHaveBeenCalled();
    expect(res.json).not.toHaveBeenCalled();
  });

  it('deve retornar status 403 e mensagem de erro se req.user não existir', () => {
    adminMiddleware(req, res, next);

    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
    expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
  });

  it('deve retornar status 403 e mensagem de erro se req.user.role não for admin', () => {
    req.user = { role: 'user' };

    adminMiddleware(req, res, next);

    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
    expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
  });

  it('deve retornar status 403 e mensagem de erro se req.user existir mas a propriedade role estiver ausente', () => {
    req.user = { name: 'John Doe' };

    adminMiddleware(req, res, next);

    expect(next).not.toHaveBeenCalled();
    expect(res.status).toHaveBeenCalledWith(403);
    expect(res.json).toHaveBeenCalledWith({ message: 'Acesso negado: Admins apenas.' });
  });
});
```
